\section{Projects}

\project{Football Match Outcome Prediction \& Data Audit}
  {\href{https://github.com/Shah-123/football-match-outcome-prediction}{GitHub} \textbar{} 2026}
  {Python, pandas, scikit-learn, Plotly}
\begin{itemize}
  \item Audited a 10-table Transfermarkt export (63K games, 1.6M appearances, 2.3M lineup rows,
        31 leagues) in pure Python; fixed silent join losses, missed second-yellow dismissals
        and future-leaking snapshot columns.
  \item Parsed own goals from free-text match events, raising scoreline reconciliation from
        91.2\% to 98.5\% of games; found that league-position columns were post-match values
        (target leakage) and excluded them.
  \item Built sequential Elo and rolling 5-match form with no look-ahead; on a time-based holdout
        of 760 Premier League matches: \textbf{48.0\% accuracy vs.\ 41.7\% baseline} (macro-F1 0.36,
        log loss 1.05).
\end{itemize}

\project{Emotion Detection from Text (NLP)}
  {\href{https://github.com/Shah-123/Emotion_detection}{GitHub} \textbar{} 2026}
  {TensorFlow/Keras, scikit-learn, FastAPI, Docker, pytest}
\begin{itemize}
  \item Built a 6-class emotion classifier on 16K labelled sentences; EDA of class imbalance
        (joy 33.5\% vs.\ surprise 3.6\%) and sentence length (95th percentile: 41 words) set the input length.
  \item RNN/LSTM baselines failed to converge; a BiLSTM with max-pooling reached
        \textbf{92.5\% test accuracy (macro-F1 0.88)}.
  \item Served it via a FastAPI endpoint and web UI in Docker; fixed a train/serve padding
        mismatch and normalised apostrophes and accents in user input, covered by 8 unit tests.
\end{itemize}
